# ============================================
# COMPOSER (PHP Dependency Manager)
# ============================================
# Install globally (all Linux distributions):

# curl -sS https://getcomposer.org/installer | php
# sudo mv composer.phar /usr/local/bin/composer
# sudo chmod +x /usr/local/bin/composer

# Verify:
# composer --version

# ============================================
# VERIFICATION COMMANDS
# ============================================

# Check installed versions:
# php -v                     # Should show 8.0 or higher
# mysql --version            # Should show MariaDB 10.5+ or MySQL 8.0+
# apache2 -v                 # Should show Apache 2.4+ (or nginx -v)
# composer --version         # Should show 2.0+
# git --version              # Should show 2.0+

# Check PHP extensions:
# php -m | grep -E 'pdo_mysql|mbstring|xml|curl|zip|gd|intl|bcmath'

# Check services:
# sudo systemctl status apache2    # or nginx
# sudo systemctl status mariadb    # or mysqld

# ============================================
# POST-INSTALLATION STEPS
# ============================================

# 1. Enable Apache modules (Ubuntu/Debian):
# sudo a2enmod rewrite ssl headers
# sudo systemctl restart apache2

# 2. Secure MariaDB/MySQL:
# sudo mysql_secure_installation

# 3. Start and enable services:
# sudo systemctl enable apache2    # or nginx
# sudo systemctl enable mariadb    # or mysqld
# sudo systemctl start apache2
# sudo systemctl start mariadb

# 4. Configure firewall:
# Ubuntu/Debian:
#   sudo ufw allow 80/tcp
#   sudo ufw allow 443/tcp
#   sudo ufw allow 22/tcp
#   sudo ufw enable
#
# CentOS/RHEL:
#   sudo firewall-cmd --permanent --add-service=http
#   sudo firewall-cmd --permanent --add-service=https
#   sudo firewall-cmd --permanent --add-service=ssh
#   sudo firewall-cmd --reload
