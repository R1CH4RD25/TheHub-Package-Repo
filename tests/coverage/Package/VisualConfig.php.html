5/**
6 * Visual Configuration System
7 *
8 * Converts JSON visual config into CSS custom properties (design tokens).
9 * Every configurable visual property uses the `--v-*` prefix so users
10 * can adjust spacing, widths, colors, density, etc., without touching CSS.
11 *
12 * Philosophy:
13 *   - Components should NEVER rely on "prayer" for layout.
14 *   - Every visual dimension is a knob users can turn.
15 *   - Presets provide sensible starting points; overrides handle the rest.
16 *   - Contributors define component definitions; admins adjust visuals.
17 *
18 * Token Naming Convention:
19 *   --v-{scope}-{property}
20 *
21 *   Scopes:
22 *     table      Table-specific tokens
23 *     filter     Filter/search bar tokens
24 *     form       Form layout tokens
25 *     card       Dashboard card tokens
26 *     page       Page-level layout tokens
27 *     text       Typography tokens
28 *
29 *   Examples:
30 *     --v-table-row-height: 48px;
31 *     --v-filter-input-width: 220px;
32 *     --v-table-cell-padding-x: 1rem;
33 *     --v-card-columns: 4;
34 *
35 * @see ComponentSchema for structural definitions
36 * @see VisualPresets for built-in presets
37 */

315    /**
316     * Sanitize a token value.
317     * Prevents CSS injection while allowing valid CSS values.
318     */
319    private static function sanitizeValue(string $value, array $tokenDef): ?string
320    {
321        // Remove any attempts at CSS injection
322        $value = trim($value);
323
324        // Block dangerous patterns
325        if (preg_match('/[;{}]|url\s*\(|expression\s*\(|javascript:|@import/i', $value)) {
326            return null;
327        }
328
329        // Type-specific validation
330        switch ($tokenDef['type']) {
331            case 'color':
332                // Accept hex, rgb(), rgba(), hsl(), hsla(), named colors
333                if (!preg_match('/^(#[0-9a-fA-F]{3,8}|rgba?\([^)]+\)|hsla?\([^)]+\)|transparent|inherit|currentColor|[a-zA-Z]+)$/', $value)) {
334                    return null;
335                }
336                break;
337
338            case 'length':
339                // Accept number + unit, calc(), var()
340                if (!preg_match('/^(-?\d+\.?\d*\s*(px|rem|em|%|vw|vh|ch|ex|pt|cm|mm|in|pc)?|auto|100%|calc\([^)]+\)|var\([^)]+\)|0)$/', $value)) {
341                    return null;
342                }
343                break;
344
345            case 'number':
346                if (!is_numeric($value)) {
347                    return null;
348                }
349                break;
350
351            case 'shadow':
352                // Accept shadow syntax or 'none'
353                if ($value !== 'none' && !preg_match('/^[\d.\-\s]+(?:px|rem|em)?\s*(?:[\d.\-\s]+(?:px|rem|em)?\s*){1,3}(?:rgba?\([^)]+\)|#[0-9a-fA-F]{3,8}|[a-zA-Z]+)(?:\s*,\s*[\d.\-\s]+(?:px|rem|em)?\s*(?:[\d.\-\s]+(?:px|rem|em)?\s*){1,3}(?:rgba?\([^)]+\)|#[0-9a-fA-F]{3,8}|[a-zA-Z]+))*$/i', $value)) {
354                    return null;
355                }
356                break;
357
358            case 'string':
359                // General string — just length check
360                if (strlen($value) > 200) {
361                    return null;
362                }
363                break;
364        }
365
366        return $value;
367    }
