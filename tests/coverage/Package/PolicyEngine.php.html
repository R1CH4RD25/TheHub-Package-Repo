540    /**

542     *
543     * Config format:
544     * "timeRestriction": {
545     *   "type": "business_hours",
546     *   "timezone": "America/Chicago",
547     *   "startHour": 7,
548     *   "endHour": 18,
549     *   "allowedDays": [1,2,3,4,5]  // Mon-Fri
550     * }
551     *
552     * @param array $restriction Time restriction config
553     *
554     * @return bool True if access is allowed right now
555     */
556    public function isWithinAllowedTime(array $restriction): bool
557    {
558        $type = $restriction['type'] ?? 'always';
559
560        if ($type === 'always') {
561            return true;
562        }
563
564        $tz = new \DateTimeZone($restriction['timezone'] ?? 'America/Chicago');
565        $now = new \DateTime('now', $tz);
566
567        if ($type === 'business_hours') {
568            $startHour = $restriction['startHour'] ?? 7;
569            $endHour = $restriction['endHour'] ?? 18;
570            $allowedDays = $restriction['allowedDays'] ?? [1, 2, 3, 4, 5]; // Mon-Fri
571
572            $currentHour = (int)$now->format('G');
573            $currentDay = (int)$now->format('N'); // 1=Mon, 7=Sun
574
575            if (!in_array($currentDay, $allowedDays)) {
576                return false;
577            }
578
579            if ($currentHour < $startHour || $currentHour >= $endHour) {
580                return false;
581            }
582
583            return true;
584        }
585
586        if ($type === 'date_range') {

588            $end = new \DateTime($restriction['endDate'] ?? '+1 year', $tz);
589            return $now >= $start && $now <= $end;

592        return true; // Unknown type defaults to allow
